// h5-agent/src/views/Home.vue

// 获取轮播图数据
const loadBanners = async () => {
  try {
    const res = await get<any[]>('/h5/banners')
    banners.value = res.data || []
  } catch (error) {
    console.error('获取轮播图失败:', error)
  }
}

// 轮播图点击处理
const handleBannerClick = (banner: any) => {
  if (banner.linkType === 'PRODUCT' && banner.linkValue) {
    router.push(`/product/${banner.linkValue}`)
  } else if (banner.linkType === 'CATEGORY' && banner.linkValue) {
    currentCategoryId.value = parseInt(banner.linkValue)
    selectedCategory.value = null
    loadProducts(true)

    // 滚动到商品列表
    setTimeout(() => {
      const productsSection = document.querySelector('.products-section')
      if (productsSection) {
        productsSection.scrollIntoView({ behavior: 'smooth', block: 'start' })
      }
    }, 300)

    Toast.success(`已为您筛选分类商品`)
  }
}

## ✅ 四、数据流转验证

### 4.1 管理后台 → 后端API → H5页面

#### 流程1: 轮播图数据流转
```
1. 管理员在管理后台创建轮播图
   POST http://39.104.58.26/api/h5/admin/banners
   {
     "title": "2026新春烟花盛宴",
     "imageUrl": "uploads/products/xxx.jpg",
     "linkType": "CATEGORY",
     "linkValue": "1",
     "sort": 1,
     "status": "ACTIVE"
   }

2. 数据存储到MySQL数据库
   INSERT INTO h5_banners (title, imageUrl, linkType, ...) VALUES (...)

3. H5页面调用公开API获取轮播图
   GET http://39.104.58.26/api/h5/banners
   WHERE status = 'ACTIVE'
   AND (startTime IS NULL OR startTime <= NOW())
   AND (endTime IS NULL OR endTime >= NOW())
   ORDER BY sort ASC

4. H5页面渲染轮播图
   <t-swiper>
     <t-swiper-item v-for="banner in banners">
       <img :src="banner.imageUrl" />
     </t-swiper-item>
   </t-swiper>
```

#### 流程2: 公告数据流转
```
1. 管理员在管理后台创建公告
   POST http://39.104.58.26/api/h5/admin/notices

2. 存储到 h5_notices 表

3. H5页面调用公开API
   GET http://39.104.58.26/api/h5/notices

4. H5页面显示跑马灯
   <t-notice-bar
     v-for="notice in notices"
     :content="notice.title"
   />
```

#### 流程3: 商品数据流转
```
1. 管理员在管理后台管理商品
   POST/PUT http://39.104.58.26/api/admin/products

2. 存储到 products 表

3. H5页面调用商品API
   GET http://39.104.58.26/api/products?page=1&pageSize=10

4. H5页面渲染商品列表
   <div class="product-card" v-for="product in products">
     <img :src="product.images[0]" />
     <div class="name">{{ product.name }}</div>
     <div class="price">¥{{ product.agentPrice }}</div>
   </div>
```

**验证结果**: ✅ **通过** - 数据流转链路完整
