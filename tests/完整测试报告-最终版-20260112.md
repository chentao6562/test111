    amount: directAmount,
    rate: level1Rate,
    type: 'DIRECT',
    status: 'PENDING'
  });
}

// 间接分润（上上级代理，仅三级代理下单时有）
if (sourceAgent.parent && sourceAgent.parent.parentId) {

    amount: indirectAmount,
    rate: level2Rate,
    type: 'INDIRECT',

  });
